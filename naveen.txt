<script>
/* ------------------------------------------------------------------ */
/* State — persisted to this browser only (localStorage)               */
/* ------------------------------------------------------------------ */
const STORE_KEY = "landslide_ews_state_v1";

const DEFAULT_STATIONS = [
  { id:"ST-01", name:"North Ridge",    location:"Sector A, Slope 34", lat:30.0668, lng:79.0193 },
  { id:"ST-02", name:"River Bend Cut", location:"Sector B, Slope 12", lat:30.0720, lng:79.0260 },
  { id:"ST-03", name:"Quarry Face",    location:"Sector C, Slope 27", lat:30.0605, lng:79.0110 },
];

function loadState(){
  try{
    const raw = localStorage.getItem(STORE_KEY);
    if (raw) return JSON.parse(raw);
  }catch(e){ /* fall through to defaults */ }
  return { stations: DEFAULT_STATIONS, readings: [], alerts: [], nextId: 1 };
}
function saveState(){
  try{ localStorage.setItem(STORE_KEY, JSON.stringify(state)); }
  catch(e){ /* storage unavailable — app still works for this session */ }
}
let state = loadState();

/* ------------------------------------------------------------------ */
/* Risk model — mirrors the Python backend's compute_risk()            */
/* ------------------------------------------------------------------ */
function normalize(v, lo, hi){
  if (hi === lo) return 0;
  return Math.max(0, Math.min(1, (v - lo) / (hi - lo)));
}

function computeRisk(rainfall, moisture, slope, pressure, vibration){
  const r = normalize(rainfall, 0, 150);
  const s = normalize(moisture, 10, 100);
  const a = normalize(slope, 5, 60);
  const p = normalize(pressure, 0, 80);
  const v = normalize(vibration, 0, 25);

  let weighted = (r*0.30) + (s*0.25) + (a*0.20) + (p*0.15) + (v*0.10);
  let score = Math.round(weighted * 1000) / 10;

  const dangerFactors = [r,s,a,p,v].filter(f => f > 0.7).length;
  if (dangerFactors >= 3) score = Math.min(100, score + 10);

  let level;
  if (score < 25) level = "Low";
  else if (score < 50) level = "Moderate";
  else if (score < 75) level = "High";
  else level = "Critical";

  const factors = {
    rainfall: Math.round(r*1000)/10,
    soil_moisture: Math.round(s*1000)/10,
    slope_angle: Math.round(a*1000)/10,
    pore_pressure: Math.round(p*1000)/10,
    vibration: Math.round(v*1000)/10,
  };
  return { score, level, factors };
}

const ALERT_THRESHOLD = 50;
const LEVEL_MESSAGE = {
  Low: "Conditions stable. Routine monitoring.",
  Moderate: "Elevated risk factors detected. Increase monitoring frequency.",
  High: "Multiple risk indicators elevated. Prepare warning protocols.",
  Critical: "Slide risk critical. Issue evacuation warning immediately.",
};
const LEVEL_COLOR = { Low:"#3FB950", Moderate:"#D4A72C", High:"#E58A2E", Critical:"#E5534B" };

function fmtTime(iso){
  return new Date(iso).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
}

function drawGauge(score, level){
  const svg = document.getElementById('gauge');
  const cx=100, cy=105, r=85;
  const pct = Math.max(0, Math.min(100, score)) / 100;
  function point(a){ return [cx + r*Math.cos(a), cy - r*Math.sin(a)]; }
  const [sx,sy] = point(Math.PI);
  const [ex,ey] = point(0);
  const angle = Math.PI - (Math.PI * pct);
  const [mx,my] = point(angle);
  const largeArc = pct > 0.5 ? 1 : 0;
  const track = `M ${sx} ${sy} A ${r} ${r} 0 1 1 ${ex} ${ey}`;
  const value  = `M ${sx} ${sy} A ${r} ${r} 0 ${largeArc} 1 ${mx} ${my}`;
  svg.innerHTML = `
    <path d="${track}" fill="none" stroke="#2B3440" stroke-width="14" stroke-linecap="round"/>
    ${score > 0 ? `<path d="${value}" fill="none" stroke="${LEVEL_COLOR[level]||'#8B96A5'}" stroke-width="14" stroke-linecap="round"/>` : ""}
  `;
}

function drawFactors(factors){
  const el = document.getElementById('factors');
  if (!factors){ el.innerHTML = `<div class="chart-empty">No readings yet.</div>`; return; }
  const labels = { rainfall:"Rainfall", soil_moisture:"Soil moisture", slope_angle:"Slope angle", pore_pressure:"Pore pressure", vibration:"Vibration" };
  el.innerHTML = Object.entries(factors).map(([k,v]) => `
    <div class="factor-row">
      <div class="label">${labels[k]}</div>
      <div class="bar"><span style="width:${v}%; background:${v>70?'#E5534B':v>40?'#D4A72C':'#3FB950'}"></span></div>
      <div class="val">${v}%</div>
    </div>`).join('');
}

function drawChart(){
  const container = document.getElementById('chart-container');
  const recent = state.readings.slice(-30);
  if (recent.length < 2){
    container.innerHTML = `<div class="chart-empty">Need at least 2 readings to draw a trend.</div>`;
    return;
  }
  const w = Math.max(container.clientWidth || 600, 500), h = 200, pad = 24;
  const stepX = (w - pad*2) / (recent.length - 1);
  const pts = recent.map((d,i) => {
    const x = pad + i*stepX;
    const y = h - pad - (d.risk_score/100)*(h-pad*2);
    return [x,y];
  });
  const path = pts.map((p,i) => (i===0?"M":"L") + p[0].toFixed(1) + " " + p[1].toFixed(1)).join(" ");
  const area = path + ` L ${pts[pts.length-1][0].toFixed(1)} ${h-pad} L ${pts[0][0].toFixed(1)} ${h-pad} Z`;
  const gridLines = [0,25,50,75,100].map(v => {
    const y = h - pad - (v/100)*(h-pad*2);
    return `<line x1="${pad}" y1="${y}" x2="${w-pad}" y2="${y}" stroke="#2B3440" stroke-width="1"/>
            <text x="2" y="${y+3}" font-size="10" fill="#8B96A5">${v}</text>`;
  }).join('');
  const dots = pts.map((p,i) => `<circle cx="${p[0]}" cy="${p[1]}" r="3" fill="${LEVEL_COLOR[recent[i].risk_level]}"/>`).join('');
  container.innerHTML = `
    <svg width="${w}" height="${h}" viewBox="0 0 ${w} ${h}">
      ${gridLines}
      <path d="${area}" fill="rgba(212,167,44,0.08)" stroke="none"/>
      <path d="${path}" fill="none" stroke="#D4A72C" stroke-width="2"/>
      ${dots}
    </svg>`;
}

function renderAlerts(){
  const body = document.getElementById('alerts-body');
  const alerts = state.alerts.slice(-15).reverse();
  if (alerts.length === 0){
    body.innerHTML = `<tr class="empty-row"><td colspan="4">No alerts triggered yet.</td></tr>`;
    return;
  }
  body.innerHTML = alerts.map(a => `
    <tr>
      <td>${fmtTime(a.timestamp)}</td>
      <td>${a.station_id}</td>
      <td><span class="badge ${a.risk_level}">${a.risk_level}</span></td>
      <td>${a.message}</td>
    </tr>`).join('');
}

function renderStations(){
  const sel = document.getElementById('station_id');
  const prev = sel.value;
  sel.innerHTML = state.stations.map(s => `<option value="${s.id}">${s.name} (${s.id})</option>`).join('');
  if (prev) sel.value = prev;

  const list = document.getElementById('station-list');
  list.innerHTML = state.stations.map(s => {
    const latest = [...state.readings].reverse().find(r => r.station_id === s.id);
    return `<li>
      <div><div class="name">${s.name}</div><span class="loc">${s.location}</span></div>
      ${latest ? `<span class="badge ${latest.risk_level}">${latest.risk_level}</span>`
                : `<span class="badge" style="background:#2B3440; color:#8B96A5;">No data</span>`}
    </li>`;
  }).join('');
}

function renderSummary(){
  let overallScore = 0, overallLevel = "Low", overallReading = null;
  state.stations.forEach(s => {
    const latest = [...state.readings].reverse().find(r => r.station_id === s.id);
    if (latest && latest.risk_score > overallScore){
      overallScore = latest.risk_score; overallLevel = latest.risk_level; overallReading = latest;
    }
  });
  drawGauge(overallScore, overallLevel);
  document.getElementById('gauge-score').textContent = `${overallScore.toFixed(1)} / 100`;
  const statusMsgs = {
    Low:"Ground is stable across all stations.",
    Moderate:"Some risk factors elevated. Keep monitoring.",
    High:"Risk elevated at one or more stations. Prepare protocols.",
    Critical:"Critical risk detected. Evacuate if directed by authorities."
  };
  document.getElementById('gauge-status').textContent = state.readings.length
    ? (statusMsgs[overallLevel] || "Awaiting data.")
    : "Submit a reading to begin monitoring.";
  drawFactors(overallReading ? overallReading.factors : null);
}

function renderAll(){
  renderStations();
  renderSummary();
  drawChart();
  renderAlerts();
  if (typeof renderStationMarkers === 'function') renderStationMarkers();
}

function showToast(level, message){
  const toast = document.getElementById('toast');
  toast.style.background = LEVEL_COLOR[level] || "#E5534B";
  toast.innerHTML = `<strong>${level} risk alert</strong>${message}`;
  toast.style.display = 'block';
  clearTimeout(showToast._t);
  showToast._t = setTimeout(() => toast.style.display = 'none', 6000);
}

document.getElementById('reading-form').addEventListener('submit', (e) => {
  e.preventDefault();
  const station_id = document.getElementById('station_id').value;
  const rainfall = parseFloat(document.getElementById('rainfall_mm').value);
  const moisture = parseFloat(document.getElementById('soil_moisture_pct').value);
  const slope = parseFloat(document.getElementById('slope_angle_deg').value);
  const pressure = parseFloat(document.getElementById('pore_pressure_kpa').value);
  const vibration = parseFloat(document.getElementById('vibration_mm_s').value);

  if ([rainfall, moisture, slope, pressure, vibration].some(v => Number.isNaN(v))){
    alert("All sensor fields must be numeric.");
    return;
  }

  const { score, level, factors } = computeRisk(rainfall, moisture, slope, pressure, vibration);
  const timestamp = new Date().toISOString();
  const reading = {
    id: state.nextId++, station_id, timestamp,
    rainfall_mm: rainfall, soil_moisture_pct: moisture, slope_angle_deg: slope,
    pore_pressure_kpa: pressure, vibration_mm_s: vibration,
    risk_score: score, risk_level: level, factors
  };
  state.readings.push(reading);

  if (score >= ALERT_THRESHOLD){
    const message = LEVEL_MESSAGE[level];
    state.alerts.push({ station_id, timestamp, risk_score: score, risk_level: level, message });
    showToast(level, message);
    if (level === "Critical") fireLocalAlert(`🚨 Critical risk — ${station_id}`, message);
  }

  saveState();
  renderAll();
});

  state = { stations: DEFAULT_STATIONS, readings: [], alerts: [], nextId: 1 };
  saveState();
  renderAll();
});

const notifDot = document.getElementById('notif-dot');
const notifLabel = document.getElementById('notif-label');
const notifBtn = document.getElementById('notif-btn');

function refreshNotifUI(){
  const supported = 'Notification' in window;
  if (!supported){
    notifLabel.textContent = 'Not supported on this browser';
    notifBtn.disabled = true;
    return;
  }
  if (Notification.permission === 'granted'){
    notifDot.classList.add('on');
    notifLabel.textContent = 'Device alerts on (this browser only)';
    notifBtn.textContent = 'Enabled';
    notifBtn.disabled = true;
  } else {
    notifDot.classList.remove('on');
    notifLabel.textContent = 'Device alerts off';
    notifBtn.textContent = 'Enable';
    notifBtn.disabled = false;
  }
}
notifBtn.addEventListener('click', async () => {
  if (!('Notification' in window)) return;
  await Notification.requestPermission();
  refreshNotifUI();
});
refreshNotifUI();

function fireLocalAlert(title, body){
  if ('Notification' in window && Notification.permission === 'granted'){
    try{ new Notification(title, { body, icon: undefined }); }catch(e){}
  }
  if ('vibrate' in navigator){
    try{ navigator.vibrate([200,100,200,100,400]); }catch(e){}
  }
}

const sosBtn = document.getElementById('sos-btn');
const sosActions = document.getElementById('sos-actions');

sosBtn.addEventListener('click', () => {
  if (!confirm("Trigger SOS? This logs a critical alert with your current location.")) return;

  const logSOS = (lat, lng) => {
    const timestamp = new Date().toISOString();
    const locText = (lat != null) ? `at ${lat.toFixed(5)}, ${lng.toFixed(5)}` : "(location unavailable)";
    const message = `Manual SOS triggered ${locText}. Immediate attention required.`;

    state.alerts.push({ station_id: "SOS", timestamp, risk_score: 100, risk_level: "Critical", message });
    saveState();
    renderAlerts();
    showToast("Critical", message);
    fireLocalAlert("🚨 SOS triggered", message);

    if (lat != null && mapInstance){
      L.marker([lat, lng], { icon: pinIcon("Critical") }).addTo(mapInstance)
        .bindPopup(`<strong>SOS</strong><br>${fmtTime(timestamp)}`).openPopup();
      mapInstance.panTo([lat, lng]);
    }

    const contact = (localStorage.getItem('landslide_ews_contact') || '').trim();
    const mapsLink = (lat != null) ? `https://maps.google.com/?q=${lat},${lng}` : "";
    const smsBody = encodeURIComponent(`SOS — landslide risk emergency. My location: ${mapsLink || 'unavailable'}`);

    if (contact){
      sosCall.href = `tel:${contact}`;
      sosText.href = `sms:${contact}?&body=${smsBody}`;
    } else {
      sosCall.href = `tel:`;
      sosText.href = `sms:?&body=${smsBody}`;
    }
    sosActions.classList.add('show');
  };

  if ('geolocation' in navigator){
    navigator.geolocation.getCurrentPosition(
      pos => logSOS(pos.coords.latitude, pos.coords.longitude),
      () => logSOS(null, null),
      { enableHighAccuracy:true, timeout:6000 }
    );
  } else {
    logSOS(null, null);
  }
});

let mapInstance = null;

function pinIcon(level){
  const color = LEVEL_COLOR[level] || "#8B96A5";
  return L.divIcon({
    className: '',
    html: `<div style="width:16px;height:16px;border-radius:50%;background:${color};border:2px solid #fff;box-shadow:0 0 6px rgba(0,0,0,.4);"></div>`,
    iconSize: [16,16], iconAnchor: [8,8]
  });
}

function initMap(){
  const el = document.getElementById('map');
  if (!el || typeof L === 'undefined') return; // Leaflet failed to load (offline first visit)

  const withCoords = state.stations.filter(s => s.lat != null && s.lng != null);
  const center = withCoords.length ? [withCoords[0].lat, withCoords[0].lng] : [20.5937, 78.9629];

  mapInstance = L.map('map', { attributionControl:false }).setView(center, withCoords.length ? 14 : 5);
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    maxZoom: 18
  }).addTo(mapInstance);
  L.control.attribution({ position:'bottomright', prefix:false }).addAttribution('© OpenStreetMap').addTo(mapInstance);

  renderStationMarkers();

  mapInstance.on('click', (e) => {
    const { lat, lng } = e.latlng;
    L.marker([lat, lng], { icon: pinIcon("Moderate") }).addTo(mapInstance)
      .bindPopup(`<strong>Incident pin</strong><br>${lat.toFixed(5)}, ${lng.toFixed(5)}`).openPopup();
  });
}

let stationMarkers = [];
function renderStationMarkers(){
  if (!mapInstance) return;
  stationMarkers.forEach(m => mapInstance.removeLayer(m));
  stationMarkers = [];
  state.stations.forEach(s => {
    if (s.lat == null || s.lng == null) return;
    const latest = [...state.readings].reverse().find(r => r.station_id === s.id);
    const level = latest ? latest.risk_level : "Low";
    const marker = L.marker([s.lat, s.lng], { icon: pinIcon(level) }).addTo(mapInstance)
      .bindPopup(`<strong>${s.name}</strong><br>${s.location}<br>Risk: ${level}`);
    stationMarkers.push(marker);
  });
}

renderAll();
initMap();
